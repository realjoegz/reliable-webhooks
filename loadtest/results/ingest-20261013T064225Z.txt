-- What became of the events of the scenario and of their deliveries, how many of the deliveries
-- were made after the last event was accepted, and how long after it the last one was.
WITH last_event AS (
  SELECT max(created_at) AS created_at FROM event WHERE created_at >= :'started'
)
SELECT (SELECT count(*) FROM event WHERE created_at >= :'started') AS events,
       count(*) AS deliveries,
       count(*) FILTER (WHERE d.state = 'succeeded') AS succeeded,
       count(*) FILTER (WHERE d.state <> 'succeeded') AS not_succeeded,
       count(*) FILTER (WHERE d.attempts > 1) AS claimed_more_than_once,
       count(*) FILTER (WHERE d.completed_at > last_event.created_at) AS made_after_the_last_event,
       round(extract(epoch FROM max(d.completed_at) - max(last_event.created_at)), 1)
         AS seconds_from_it_to_the_last_delivery
FROM delivery d
CROSS JOIN last_event
WHERE d.created_at >= :'started';

-- For each two minutes from the first event on, which is a step of loadtest/ingest.js: the
-- deliveries that were created in it, and how long they took from their creation to the
-- transaction that recorded their 2xx.
SELECT floor(extract(epoch FROM d.created_at - first.created_at) / 120) + 1 AS step,
       count(*) AS deliveries,
       round(percentile_cont(0.5) WITHIN GROUP (
         ORDER BY extract(epoch FROM d.completed_at - d.created_at))::numeric, 3) AS p50_seconds,
       round(percentile_cont(0.99) WITHIN GROUP (
         ORDER BY extract(epoch FROM d.completed_at - d.created_at))::numeric, 3) AS p99_seconds,
       round(max(extract(epoch FROM d.completed_at - d.created_at)), 3) AS max_seconds
FROM delivery d
CROSS JOIN (SELECT min(created_at) AS created_at FROM delivery WHERE created_at >= :'started') first
WHERE d.created_at >= :'started' AND d.state = 'succeeded'
GROUP BY step
ORDER BY step;
